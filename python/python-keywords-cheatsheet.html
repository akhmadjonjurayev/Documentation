def javob(kod):
    match kod:
        case 200:
            return "OK"
        case 401 | 403:              # yoki

        case int() if kod >= 500:    # tur + shart
            return "Server xatosi"
        case _:                      # default
            return "Noma'lum"

yield
C#yield return / IEnumerable

Funksiyani generatorga aylantiradi: qiymatlar so'ralganda bittadan hisoblanadi — katta ma'lumotda xotira tejaladi.

next(gen) — keyingi qiymat. Generator bir marta aylanadi.

def kvadratlar(n):
    for i in range(n):
        yield i * i          # qiymatni birma-bir beradi
print(list(kvadratlar(4)))   # → [0, 1, 4, 9]
gen = kvadratlar(3)
print(next(gen), next(gen))  # → 0 1
globalnonlocal
C#static maydon / closure (C# da avtomatik)

Funksiya ichida tashqaridagi o'zgaruvchiga yangi qiymat berish uchun. Faqat o'qish uchun kerak emas.

global — modul (fayl) darajasidagi; nonlocal — o'rab turgan funksiyadagi. global dan kam foydalaning.

hisob = 0
def oshir():
    global hisob             # modul darajasidagi o'zgaruvchi
    hisob += 1
oshir()
print(hisob)                 # → 1

def hisoblagich():
    n = 0
    def keyingi():
        nonlocal n           # tashqi funksiyadagi n
        n += 1
        return n
    return keyingi
k = hisoblagich()
print(k(), k(), k())         # → 1 2 3

6 Klasslar (OOP) class · del · self · super() · @property

class__init__self
C#class, konstruktor, this, static maydon

__init__ — konstruktor. self — obyektning o'zi (kalit so'z emas, lekin doim 1-parametr). Klass ichida to'g'ridan-to'g'ri yozilgan o'zgaruvchi — hamma obyektlar uchun umumiy.

Obyekt yaratish: Xodim("Ali") — new yo'q. Konstruktorsiz bo'sh klass: class A: pass.

class Xodim:
    kompaniya = "Tech"                 # klass (static) maydon
    def __init__(self, ism, maosh=0):  # konstruktor
        self.ism = ism                 # self = this
        self.maosh = maosh
    def tanishtir(self):               # self — doim 1-parametr
        return f"{self.ism}, {self.kompaniya}"

ali = Xodim("Ali", 15_000_000)         # new yozilmaydi
print(ali.tanishtir())                 # → Ali, Tech
print(ali.maosh, Xodim.kompaniya)      # → 15000000 Tech

C#class B : A, base(…), override, is

Ota klass qavs ichida yoziladi. super() — ota klass metodini chaqirish. virtual/override so'zlari yo'q — bir xil nom bilan qayta yozilsa yetarli.

class Menejer(Xodim):                  # C#: : Xodim
    def __init__(self, ism, jamoa):
        super().__init__(ism)          # C#: base(ism)
        self.jamoa = jamoa

@property@x.setter@classmethod@staticmethod
C#get / set, static metod

@property — metodni maydon kabi o'qish (getter), @x.setter — qiymatni tekshirib yozish. @classmethod — cls (klassning o'zi) bilan ishlaydi, @staticmethod — na self, na cls.

_nom — «ichki, tegmang» kelishuvi; __nom — nomi yashiriladi (private'ga yaqin). @classmethod ko'pincha muqobil konstruktor: Xodim.from_string("Ali,1500").

class Hisob:
    soni = 0
    def __init__(self, balans):
        self.balans = balans           # setter ishlaydi
        Hisob.soni += 1
    @property                          # C#: get
    def balans(self):
        return self._balans
    @balans.setter                     # C#: set
    def balans(self, qiymat):
        if qiymat < 0:
            raise ValueError("Manfiy bo'lmaydi")
        self._balans = qiymat
    @classmethod                       # cls = klassning o'zi
    def nechta(cls):
        return cls.soni
    @staticmethod                      # self ham, cls ham yo'q
    def valyuta():
        return "UZS"

h = Hisob(100)
h.balans += 50
print(h.balans, Hisob.nechta(), Hisob.valyuta())  # → 150 1 UZS

C#ToString(), indexer, operator +

class Savat:
    def __init__(self, *narsalar):
        self.narsalar = list(narsalar)
    def __repr__(self):                  # print(s)
        return f"Savat{self.narsalar}"
    def __len__(self):                   # len(s)
        return len(self.narsalar)
    def __getitem__(self, i):            # s[i]
        return self.narsalar[i]
    def __add__(self, boshqa):           # s1 + s2
        return Savat(*self.narsalar, *boshqa.narsalar)

s = Savat("olma") + Savat("non", "sut")
print(s)                    # → Savat['olma', 'non', 'sut']
print(len(s), s[1])         # → 3 non

7 Xatolar bilan ishlash try · except · else · finally · raise · assert

tryexceptelsefinally
C#try / catch / — / finally

try — xato chiqishi mumkin bo'lgan kod. except Tur as e — xatoni ushlash. else — xato bo'lmasa. finally — har doim.

Aniq xato turini ushlang. Hammasini: except Exception as e: (yalang'och except: tavsiya etilmaydi).

def bol(a, b):
    try:
        natija = a / b
    except ZeroDivisionError:          # C#: catch (...)
        print("0 ga bo'lib bo'lmaydi")
    except (TypeError, ValueError) as e:
        print("Xato:", e)              # e — xato obyekti
    else:                              # xato BO'LMASA
        print("Natija:", natija)
    finally:                           # HAR DOIM
        print("tugadi")

bol(10, 4)     # → Natija: 2.5, tugadi
bol(1, 0)      # → 0 ga bo'lib bo'lmaydi, tugadi
