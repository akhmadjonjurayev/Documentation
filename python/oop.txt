        self.name = name
        self.salary = salary

    @classmethod
    def from_string(cls, text):
        name, salary = text.split(",")
        return cls(name, int(salary))

class Employee:
    company_name = "Tech Company"
    employee_count = 0

    def __init__(self, name, salary):
        self.name = name
        self.salary = salary
        Employee.employee_count += 1

    @property
    def salary(self):
        return self.__salary

    @salary.setter
    def salary(self, value):
        if value < 0:
            raise ValueError("Salary cannot be negative")

        self.__salary = value

    def describe(self):
        return (
            f"{self.name} works at {self.company_name}. "
            f"Salary: {self.salary}"
        )

    @staticmethod
    def is_valid_salary(value):
        return value >= 0

    @classmethod
    def get_employee_count(cls):
        return cls.employee_count


ali = Employee("Ali", 15000000)
vali = Employee("Vali", 12000000)

print(ali.describe())
print(vali.describe())

print(Employee.get_employee_count())
print(Employee.is_valid_salary(5000000))
